const source = $input.item.json;

const systemPrompt = `You are a product capability analyst for Ping Identity. Document Ping's actual, verifiable product capabilities from official sources.

SOURCE-SPECIFIC EXTRACTION GUIDANCE (how to read each source type):

documentation:
Look for: clear instructions, scope description, parameter references, operational guidance, and production configuration details.
Maturity bias: documented by default. production_mature is supportable only when the page clearly indicates real production support, operational readiness, or broad productized usage. Do not assume broad maturity from a single narrow doc page.

release_notes:
Look for: version numbers, dates, feature names, GA/EA labels, and rollout language. These are strong evidence of shipped capability. If a feature appears in GA release notes with a version and date, production_mature is often supportable — but only for the capability explicitly described. Extract the specific version/date as the observed_fact.

developer_portal:
Look for: API references, SDK usage examples, getting-started guides, and implementation guidance.
Maturity bias: documented. Do not assign production_mature from portal pages alone unless the page explicitly includes stronger evidence of shipped and operational support.

github_repo:
Look for: release tags, versioned README, explicit usage instructions, structured setup guidance, changelog, active version history, and clear feature support statements.
Maturity bias: partial, documented, or unclear by default. production_mature is only supportable when the repo itself provides strong evidence of maintained, shipped, production-oriented support. Repo existence alone = unclear. Code presence alone is not enough.

sample_app:
Treat as weak evidence. The capability is demonstrable but not production-supported. Always assign sample_only. Do not infer production maturity from demo, reference, or sample code.

CRITICAL RULES — DO NOT VIOLATE:
1. Code in a repository is NOT evidence of a production capability unless the README has explicit usage instructions and version/release history.
2. A sample app means a demo exists, not that the capability is mature. Never assign production_mature from a sample_app source.
3. One documentation page does NOT mean a complete experience. Scope maturity only to what this specific page covers.
4. Absence of a public artifact does NOT mean the capability is missing. Do not make absence claims.
5. Marketing language (seamless, powerful, AI-native) is NOT evidence. Ignore it.
6. If content is vague, too short, or lacks actionable detail — return [].
7. source_weight is supporting context for your judgment, not an automatic maturity override.

SUPPORT MATURITY VOCABULARY (use exactly these values):
- production_mature: GA capability with version history, documented on at least one clear official page with release evidence. Achievable from one strong source if the evidence is clear and unambiguous.
- documented: Official doc or developer portal page with clear, actionable instructions.
- partial: Documentation explicitly notes limited platform coverage, partial scope, or incomplete features.
- beta_preview: Explicitly labeled beta, preview, experimental, or early access in the source.
- sample_only: The only evidence is a sample, demo, or reference implementation with no official doc or release note.
- unclear: Source has content but maturity cannot be reliably determined from it.

CAPABILITY TAGS — assign exactly one. Prefer the default tag when it clearly matches. Override only if the content maps more precisely to a different tag:
mcp, sdk, sample_apps, platform_ui, end_user_ui, hosted_pages, hosted_account_pages, localization, customization, login_widget, javascript_sdk, android_sdk, ios_sdk, flutter, astro, developer_portal, aic_dev_experience, release_notes, docs

CONFIDENCE: Only include capabilities with confidence >= 3. Return [] if nothing qualifies. Output raw JSON only — no markdown fences, no commentary.`;

Apply evidence rules based on source_type above.\nUse the default capability tag unless the content clearly maps to a different one.\n\nReturn a JSON array:\n[{"feature_name":"","capability_tag":"","support_maturity":"documented","maturity_reason":"one sentence citing specific evidence from the content","observed_fact":"","evidence_summary":"","confidence":3}]\n\nIf nothing meets the confidence threshold, return [].`;

## Task 2: Update `Parse Capabilities` node

Add 3 parser safety cap rules. Rules are injected after the confidence filter and before the field mapping block.

**New `jsCode` for `Parse Capabilities`:**

```javascript
const source = $('Build Sonnet Prompt').item.json;
const apiResponse = $input.item.json;

function deriveSourceWeight(sourceType) {
  const w = {
    documentation: 'high',
    release_notes: 'medium-high',
    developer_portal: 'medium',
    github_repo: 'medium',
    product_page: 'medium',
    sample_app: 'low'
  };
  return w[sourceType] || 'low';
}

let capabilities = [];
try {
  const text = apiResponse.content?.[0]?.text || '[]';
  const match = text.match(/\[[\s\S]*\]/);
  if (match) capabilities = JSON.parse(match[0]);
  if (!Array.isArray(capabilities)) capabilities = [];
} catch(e) { capabilities = []; }

capabilities = capabilities.filter(c => {
  const conf = c.confidence !== undefined ? c.confidence : (c.confidence_score || 0);
  return conf >= 3;
});

if (capabilities.length === 0) return { json: { _no_caps: true } };

// Stage 2: Parser safety caps
const STRONG_SIGNALS = [
  'release', 'releases', 'version', 'tag', 'v1', 'v2', 'v3',
  'changelog', 'ga', 'generally available', 'stable', 'latest',
  'published', 'documentation', 'docs'
];
const USAGE_SIGNALS = [
  'install', 'setup', 'usage', 'guide', 'example',
  'how to', 'getting started', 'readme'
];

capabilities = capabilities.map(cap => {
  const conf = cap.confidence !== undefined ? cap.confidence : (cap.confidence_score || 3);
  const maturity = cap.support_maturity || cap.support_level || 'unclear';
  const evidenceText = (
    (cap.evidence_summary || cap.summary || '') + ' ' +
    (cap.observed_fact || '')
  ).toLowerCase();

  let cappedMaturity = maturity;
  let cappedConfidence = conf;
  let parserCapped = false;

  // Rule 1: sample_app hard cap
  if (source.source_type === 'sample_app') {
    if (cappedMaturity !== 'sample_only') {
      cappedMaturity = 'sample_only';
      parserCapped = true;
    }
    if (cappedConfidence > 3) {
      cappedConfidence = 3;
      parserCapped = true;
    }
  }

  // Rule 2: github_repo production_mature block
  if (source.source_type === 'github_repo' && cappedMaturity === 'production_mature') {
    const hasStrongSignal = STRONG_SIGNALS.some(s => evidenceText.includes(s));
    if (!hasStrongSignal) {
      const hasUsageSignal = USAGE_SIGNALS.some(s => evidenceText.includes(s));
      cappedMaturity = hasUsageSignal ? 'documented' : 'partial';
      parserCapped = true;
    }
  }

  // Rule 3: vague evidence confidence cap
  const evidenceSummary = (cap.evidence_summary || cap.summary || '').trim();
  if (evidenceSummary.length < 40 && cappedConfidence > 3) {
    cappedConfidence = 3;
    parserCapped = true;
  }

  return {
    ...cap,
    support_maturity: cappedMaturity,
    confidence: cappedConfidence,
    maturity_reason: parserCapped
      ? `[parser-cap] ${cap.maturity_reason || ''}`
      : (cap.maturity_reason || '')
  };
});

const now = new Date().toISOString();
const sourceWeight = deriveSourceWeight(source.source_type);

return { json: { _capabilities: capabilities.map(cap => {
  const confidence = cap.confidence;
  const capabilityTag = (cap.capability_tag && cap.capability_tag.trim())
    ? cap.capability_tag.trim()
    : (source.capability_tag || '');
  return {
    capability_id:    `PING-${source.source_id}-${Date.now()}-${Math.random().toString(36).substr(2,4)}`,
    capability_area:  source.capability_area,
    feature_name:     cap.feature_name || '',
    capability_tag:   capabilityTag,
    product_name:     source.product_name,
    source_url:       source.url,
    source_id:        source.source_id,
    source_type:      source.source_type,
    source_weight:    sourceWeight,
    observed_fact:    cap.observed_fact || '',
    evidence_summary: cap.evidence_summary || cap.summary || '',
    support_maturity: cap.support_maturity,
    maturity_reason:  cap.maturity_reason || '',
    confidence:       confidence,
    last_checked:     now,
    freshness_status: 'fresh',
    week:             source.week,
    run_id:           source.run_id,
    notes:            ''
  };
}) } };
```
